################ 3. ROOT CAUSE ################
# The password was encoded with 'echo "secretpassword" | base64'. echo appends a newline (0x0a), so the Secret
# stores 'secretpassword\n' (15 bytes, base64 ending in 'K' instead of '='). The app sends that exact string,
# Postgres compares it with 'secretpassword' and rejects it: password authentication failed.
################ 4. FIX ################
$ diff 05-troubleshooting/broken-secret.yaml 05-troubleshooting/fixed-secret.yaml
1,2c1,2
< # BROKEN: the password was encoded with plain `echo "secretpassword" | base64`.
< # echo appends a newline, so the stored value is "secretpassword\n" (15 bytes, base64 ends in "K").
---
> # FIXED: values encoded with `echo -n ... | base64` (no trailing newline).
> # Even better: use stringData, or `kubectl create secret generic --from-literal=...`, so nobody base64-encodes by hand.
12,13c12,13
<   # echo "secretpassword" | base64      <-- missing -n
<   POSTGRES_PASSWORD: c2VjcmV0cGFzc3dvcmQK
---
>   # echo -n "secretpassword" | base64
>   POSTGRES_PASSWORD: c2VjcmV0cGFzc3dvcmQ=
