# ---- Create the Secret from the command line (values never written to a file in Git) ----
$ kubectl -n s12 create secret generic db-credentials --from-literal=DB_USER=appuser --from-literal=DB_PASSWORD='Demo-P@ss-2026'
secret/db-credentials created

$ kubectl -n s12 get secret db-credentials
NAME             TYPE     DATA   AGE
db-credentials   Opaque   2      0s

# describe hides the values, shows only sizes:
$ kubectl -n s12 describe secret db-credentials | sed -n '/^Data/,$p'
Data
====
DB_PASSWORD:  14 bytes
DB_USER:      7 bytes

# but the stored data is only base64-ENCODED, not encrypted:
$ kubectl -n s12 get secret db-credentials -o jsonpath='{.data}'; echo
{"DB_PASSWORD":"RGVtby1QQHNzLTIwMjY=","DB_USER":"YXBwdXNlcg=="}

$ kubectl -n s12 get secret db-credentials -o jsonpath='{.data.DB_PASSWORD}' | base64 -d; echo
Demo-P@ss-2026

# ---- Inject into a Pod ----
$ kubectl apply -f 02-secret/pod.yaml
pod/secret-demo created

$ kubectl -n s12 wait --for=condition=Ready pod/secret-demo --timeout=120s
pod/secret-demo condition met

# ---- Verify inside the container ----
$ kubectl -n s12 exec secret-demo -- sh -c "echo DB_USER=\$DB_USER; echo DB_PASSWORD=\$DB_PASSWORD"
DB_USER=appuser
DB_PASSWORD=Demo-P@ss-2026

$ kubectl -n s12 exec secret-demo -- ls -lL /etc/secrets
total 8
-r--------    1 root     root            14 Oct  7 06:52 DB_PASSWORD
-r--------    1 root     root             7 Oct  7 06:52 DB_USER

$ kubectl -n s12 exec secret-demo -- cat /etc/secrets/DB_PASSWORD; echo
Demo-P@ss-2026

$ kubectl -n s12 exec secret-demo -- df -h /etc/secrets   # tmpfs: kept in memory, never on node disk
Filesystem                Size      Used Available Use% Mounted on
tmpfs                     4.8G      8.0K      4.8G   0% /etc/secrets

# ---- Why a Secret YAML must not be committed: anyone with the repo can decode it ----
$ echo 'DB_PASSWORD: RGVtby1QQHNzLTIwMjY=' | awk '{print $2}' | base64 -d; echo
Demo-P@ss-2026

$ cat 02-secret/secret.example.yaml | grep -A3 stringData   # only a placeholder file is in Git
stringData:                     # plain text here; Kubernetes stores it base64-encoded in .data
  DB_USER: "appuser"
  DB_PASSWORD: "change-me"
