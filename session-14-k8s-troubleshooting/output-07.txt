# 4. FIX: use the FQDN inventory.s14-backend.svc.cluster.local (Pod env cannot be edited in place -> recreate)
$ diff scenarios/07-dns/broken.yaml scenarios/07-dns/fixed.yaml
35c35
<         - { name: INVENTORY_URL, value: "http://inventory" }     # short name only resolves in the SAME namespace
---
>         - { name: INVENTORY_URL, value: "http://inventory.s14-backend.svc.cluster.local" }
