# Normal Service -> DNS returns ONE virtual IP:
$ kubectl -n s11 exec client -- dig +short web-clusterip.s11.svc.cluster.local
10.97.245.215

# Headless Service -> DNS returns EVERY Pod IP:
$ kubectl -n s11 exec client -- dig +short db-headless.s11.svc.cluster.local
10.244.0.104
10.244.0.103
10.244.0.105

# ...and each StatefulSet Pod gets its own stable DNS name:
$ kubectl -n s11 exec client -- sh -c 'for i in 0 1 2; do echo -n "db-$i -> "; dig +short db-$i.db-headless.s11.svc.cluster.local; done'
db-0 -> 10.244.0.103
db-1 -> 10.244.0.104
db-2 -> 10.244.0.105

$ kubectl -n s11 exec client -- sh -c 'for i in 0 1 2; do curl -s http://db-$i.db-headless:5678; done'
hello from db-0
hello from db-1
hello from db-2
