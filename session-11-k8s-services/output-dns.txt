search s11.svc.cluster.local svc.cluster.local cluster.local

# Short name works only inside the same namespace (search list adds s11.svc.cluster.local)
$ kubectl -n s11 exec client -- nslookup web-clusterip | tail -3
Name:	web-clusterip.s11.svc.cluster.local
Address: 10.97.245.215


$ kubectl -n s11 exec client -- dig +search +short web-clusterip.s11
10.97.245.215

$ kubectl -n s11 exec client -- dig +search +short web-clusterip.s11.svc
10.97.245.215

$ kubectl -n s11 exec client -- dig +short web-clusterip.s11.svc.cluster.local
10.97.245.215

# A Service in ANOTHER namespace needs at least <svc>.<namespace>:
$ kubectl -n s11 exec client -- dig +search +short kubernetes          # -> kubernetes.s11.svc.cluster.local: does not exist

$ kubectl -n s11 exec client -- dig +search +short kubernetes.default
10.96.0.1

$ kubectl -n s11 exec client -- dig +short kubernetes.default.svc.cluster.local
10.96.0.1

$ kubectl -n s11 exec client -- dig +short kube-dns.kube-system.svc.cluster.local
10.96.0.10

# SRV record (port discovery) and Pod A record (dashed IP):

$ kubectl -n s11 exec client -- dig +short 10-244-0-99.s11.pod.cluster.local

# Reverse lookup of the ClusterIP
$ kubectl -n s11 exec client -- dig +short -x 10.97.245.215
web-clusterip.s11.svc.cluster.local.

# Pod-to-Service across namespaces: a Pod in 'default' calling the s11 Service
